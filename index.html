Solicitudes y cotizaciones de inventario

Procesa solicitudes: la app busca en tu base de Google Sheets el código cruce, el stock, el precio y el RUT a cotizar de cada código.

Base de datos (Google Sheets en vivo)

La app lee directamente tu Google Sheet cada vez que se abre y antes de procesar una solicitud, así que siempre usa los datos actuales. Las pestañas deben llamarse Cruce_Codigos, Inventario, Historico_Precios y Proveedores.

function doGet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var nombres = ["Cruce_Codigos", "Inventario", "Historico_Precios", "Proveedores"];
  var salida = {};
  nombres.forEach(function (n) {
    var hoja = ss.getSheetByName(n);
    salida[n] = hoja ? hoja.getDataRange().getDisplayValues() : null;
  });
  return ContentService.createTextOutput(JSON.stringify(salida))
    .setMimeType(ContentService.MimeType.JSON);
}

// ===== CONFIGURACION =====
var HOJA_CRUCE = "Cruce_Codigos";
var FF = "https://www.filterfinderonline.com";
var FG = "https://www.fleetguard.com/es";
var PREFIJOS_FG = "AF|LF|FF|FS|HF|WF|FK|AH|CV|DZ|ST|SP|UF|CC|AS|FH|FA|DA|WA|AA|PA";
var RE_FG = new RegExp("(?:^|[^A-Z0-9])((?:" + PREFIJOS_FG + ")\\d{3,6}[A-Z]{0,3})(?![A-Z0-9])", "gi");

function claveCodigo_(s) {
  return String(s == null ? "" : s).toUpperCase().replace(/[\s\-_.\/\u00a0]/g, "");
}
function respuesta_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
function traer_(url) {
  try {
    var r = UrlFetchApp.fetch(url, {
      muteHttpExceptions: true, followRedirects: true,
      headers: { "Accept": "application/json,text/html", "User-Agent": "Mozilla/5.0 (compatible; CruceBot)" }
    });
    return r.getResponseCode() === 200 ? r.getContentText() : "";
  } catch (err) { return ""; }
}
// Todos los codigos con formato Fleetguard dentro de un texto
function codigosFG_(texto) {
  var out = [], m, t = String(texto || "").toUpperCase().replace(/-/g, " ");
  RE_FG.lastIndex = 0;
  while ((m = RE_FG.exec(t)) !== null) { if (out.indexOf(m[1]) < 0) out.push(m[1]); }
  return out;
}

// ===== FUENTE 1: FilterFinder (tienda Shopify) =====
// Paso 1: busca candidatos. Paso 2: abre la ficha de cada candidato y SOLO acepta
// el cruce si el codigo cotizado aparece escrito en esa ficha (evita falsos positivos).
function handlesFF_(codigo) {
  var handles = [];
  function add(h) { if (h && handles.indexOf(h) < 0 && handles.length < 5) handles.push(h); }
  var q = encodeURIComponent(codigo);
  var js = traer_(FF + "/search/suggest.json?q=" + q + "&resources[type]=product&resources[limit]=10");
  try {
    var prods = ((JSON.parse(js).resources || {}).results || {}).products || [];
    prods.forEach(function (p) {
      var t = (p.title || "") + " " + (p.vendor || "") + " " + (p.handle || "");
      if (/fleetguard/i.test(t) || codigosFG_(t).length) add(p.handle);
    });
  } catch (e) {}
  if (!handles.length) {
    var html = traer_(FF + "/search?type=product&q=" + q);
    var re = /\/products\/([a-z0-9][a-z0-9\-]*)/gi, m;
    while ((m = re.exec(html)) !== null) { if (/fleetguard/i.test(m[1]) || codigosFG_(m[1]).length) add(m[1]); }
  }
  return handles;
}
function buscarEnFilterFinder_(codigo) {
  var clave = claveCodigo_(codigo), candidato = "";
  var hs = handlesFF_(codigo);
  for (var i = 0; i < hs.length; i++) {
    var ficha = traer_(FF + "/products/" + hs[i] + ".js");
    if (!ficha) continue;
    var p; try { p = JSON.parse(ficha); } catch (e) { continue; }
    var titulo = (p.title || "") + " " + hs[i];
    var fgs = codigosFG_(titulo);
    if (!fgs.length) continue;
    var pajar = claveCodigo_((p.title || "") + " " + (p.description || "").replace(/<[^>]+>/g, " ") + " " + (p.tags || []).join(" "));
    if (clave.length >= 4 && pajar.indexOf(clave) >= 0) return { cruce: fgs[0], fuente: "FilterFinder" };
    if (!candidato) candidato = fgs[0];
  }
  return candidato ? { candidato: candidato } : null;
}

// ===== FUENTE 2: Fleetguard.com (sitio Salesforce; mejor esfuerzo) =====
// Ese sitio dibuja los resultados con JavaScript, por lo que a veces no entrega nada al script.
function buscarEnFleetguard_(codigo) {
  var clave = claveCodigo_(codigo);
  var urls = [FG + "/search?term=" + encodeURIComponent(codigo), FG + "/global-search/" + encodeURIComponent(codigo)];
  for (var i = 0; i < urls.length; i++) {
    var html = traer_(urls[i]);
    if (!html) continue;
    var plano = html.replace(/<script[\s\S]*?<\/script>/gi, " ").replace(/<[^>]+>/g, " ");
    if (clave.length >= 4 && claveCodigo_(plano).indexOf(clave) >= 0) {
      var fgs = codigosFG_(plano);
      if (fgs.length) return { cruce: fgs[0], fuente: "Fleetguard.com" };
    }
  }
  return null;
}

function buscarEquivalente_(codigo) {
  var cand = "";
  var a = buscarEnFilterFinder_(codigo);
  if (a && a.cruce) return a;
  if (a && a.candidato) cand = a.candidato;
  var b = buscarEnFleetguard_(codigo);
  if (b && b.cruce) return b;
  return cand ? { candidato: cand } : null;
}
function nota_(eq, fecha) {
  if (eq && eq.cruce) return "Auto " + eq.fuente + " " + fecha;
  if (eq && eq.candidato) return "Auto sin confirmar (posible " + eq.candidato + ") " + fecha;
  return "Auto sin resultado " + fecha;
}
function fecha_() { return Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd-MM-yyyy"); }

// ===== REGISTRO AUTOMATICO desde la pagina =====
// El codigo SIEMPRE se agrega a Cruce_Codigos; si la busqueda falla queda como "Sin cruce" para reintentar.
function doPost(e) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
    var req = JSON.parse(e.postData.contents);
    if (req.accion !== "registrar" || !Array.isArray(req.codigos)) {
      return respuesta_({ ok: false, error: "Solicitud no valida" });
    }
    var hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJA_CRUCE);
    if (!hoja) return respuesta_({ ok: false, error: "No existe la hoja " + HOJA_CRUCE });

    var existentes = {};
    hoja.getDataRange().getValues().forEach(function (f, i) {
      if (i > 0) { existentes[claveCodigo_(f[0])] = true; if (f[1]) existentes[claveCodigo_(f[1])] = true; }
    });

    var fecha = fecha_();
    var resultados = [], filas = [];
    req.codigos.slice(0, 10).forEach(function (c) {
      c = String(c).trim();
      var kc = claveCodigo_(c);
      if (!kc || c.length > 40 || !/^[A-Za-z0-9 \-_.\/]+$/.test(c)) return;
      if (existentes[kc]) { resultados.push({ original: c, cruce: "", estado: "existente" }); return; }
      existentes[kc] = true;
      var eq = null;
      try { eq = buscarEquivalente_(c); } catch (err) {}
      filas.push([c, eq && eq.cruce ? eq.cruce : "Sin cruce", nota_(eq, fecha)]);
      resultados.push({ original: c, cruce: eq && eq.cruce ? eq.cruce : "", estado: "nuevo" });
    });

    if (filas.length) {
      if (!hoja.getRange(1, 3).getValue()) hoja.getRange(1, 3).setValue("Origen");
      hoja.getRange(hoja.getLastRow() + 1, 1, filas.length, 3).setNumberFormat("@").setValues(filas);
    }
    return respuesta_({ ok: true, resultados: resultados });
  } catch (err) {
    return respuesta_({ ok: false, error: String(err) });
  } finally {
    try { lock.releaseLock(); } catch (x) {}
  }
}

// ===== REINTENTAR los "Sin cruce" (manual desde el menu, o con activador diario) =====
function completarPendientes() {
  var hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJA_CRUCE);
  if (!hoja) return;
  var datos = hoja.getDataRange().getValues(), inicio = Date.now(), fecha = fecha_(), n = 0;
  for (var i = 1; i < datos.length; i++) {
    if (Date.now() - inicio > 270000) break; // limite de tiempo de Apps Script
    var cod = String(datos[i][0] || "").trim(), cruce = String(datos[i][1] || "").trim();
    var origen = String(datos[i][2] || "");
    if (!cod || (cruce && !/^sin cruce$/i.test(cruce))) continue;
    if (/Auto sin (resultado|confirmar)/.test(origen) && origen.indexOf(fecha) >= 0) continue; // ya se intento hoy
    var eq = null;
    try { eq = buscarEquivalente_(cod); } catch (err) {}
    if (eq && eq.cruce) { hoja.getRange(i + 1, 2).setNumberFormat("@").setValue(eq.cruce); n++; }
    hoja.getRange(i + 1, 3).setValue(nota_(eq, fecha));
  }
  try { SpreadsheetApp.getActive().toast(n + " cruce(s) nuevo(s) encontrado(s)."); } catch (x) {}
}
function onOpen() {
  SpreadsheetApp.getUi().createMenu("Cruces").addItem("Completar cruces pendientes", "completarPendientes").addToUi();
}
